**调用者如果是内部类的对象的话，使用外部类得写成外部类类名.this.成员对象，如果show方法在外部类中则需要在形参处输入内部类对象**



```java
//Car类
class Car{
    private String carName;
    private String carColor;
    private int carAge;
    class Engine{
        private String engineName;
        public void setEnginename(String enginename){
            this.engineName = enginename;
        }
    }
    public void show(Engine engine){
        System.out.println(carName);
        System.out.println(carAge);
        System.out.println(carColor);
        System.out.println(engine.engineName);
    }

//main
    public static void main(String[] args) {
     Car car =  new Car("车名","颜色",10);
     Car.Engine engine = car.new Engine();
     engine.setEnginename("V8双涡轮增压");
        car.show(engine);
    }
```



```java
class Engine{
    String engineName;
    int engineAge;
    
    public void show(){
        // === 修正图片错误 ===
        System.out.println("引擎名称：" + this.engineName);  // 访问内部类成员
        
        // ✅ 关键：使用外部类名.this访问外部类成员
        System.out.println("所属车辆：" + Car.this.carName); 
        System.out.println("车辆颜色：" + Car.this.carColor);
    }
}
```
